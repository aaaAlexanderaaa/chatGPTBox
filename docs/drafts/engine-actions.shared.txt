// Generated into the four Gemini actions. Version 3 keeps legacy ChatGPT actions
// from interpreting these engine-run documents as ChatGPT conversation controls.
const ENGINE = 'geminiweb'
const HEADER = /^<!-- chatgptbox-document:(\{[^\n]*\}) -->\n?/
const originalContent = draft.content || ''
const identifier = () => Date.now().toString(36) + Math.random().toString(36).slice(2).padEnd(12,'0') + Math.random().toString(36).slice(2).padEnd(12,'0')
const encode = (value) => JSON.stringify(value).replace(/</g, '\\u003c').replace(/>/g, '\\u003e')
const safeTitle = (value) => String(value || 'Gemini conversation').replace(/[\r\n<>]/g, ' ')
const userText = (value) => String(value).replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;')
const clone = (value) => JSON.parse(JSON.stringify(value))

function request(path, method = 'GET', data, key) {
  const response = HTTP.create().request({ url: BASE_URL + path, method,
    headers: { Authorization: 'Bearer ' + API_TOKEN, ...(key ? { 'Idempotency-Key': key } : {}) },
    ...(data === undefined ? {} : { data, encoding: 'json' }),
  })
  const body = response.responseData
  if (!response.success || response.statusCode >= 400) {
    const error = new Error(body?.error?.message || response.error || 'Gateway request could not be confirmed. Keep this note and use Get.')
    error.detail = body
    throw error
  }
  if (!body || typeof body !== 'object') throw new Error('The gateway returned an invalid result. Keep this note and use Get.')
  return body
}
function readDocument() {
  const match = originalContent.match(HEADER)
  if (!match) {
    const lines = originalContent.trim().split('\n').filter((line) => line.trim())
    const legacyList = lines.length > 0 && lines.every((line) => /^- \[[ xX]\].*<!-- chatgptbox-conversation:[A-Za-z0-9-]+ -->\s*$/.test(line))
    if (/^<!-- chatgptbox-document:/.test(originalContent) || legacyList)
      throw new Error('Use the matching ChatGPT actions for an old ChatGPT note. Gemini actions preserve it unchanged.')
    return { meta: { version: 3, engineId: ENGINE, id: identifier(), kind: 'new' }, body: originalContent }
  }
  let meta
  try { meta = JSON.parse(match[1]) } catch { throw new Error('Damaged ChatGPTBox document header. The complete note was kept.') }
  if (meta.version !== 3 || meta.engineId !== ENGINE || !/^[a-z0-9]{16,80}$/.test(meta.id) || !['new','list','conversation'].includes(meta.kind))
    throw new Error('This is not a supported Gemini document. Use its matching actions; the complete note was kept.')
  return { meta, body: originalContent.slice(match[0].length) }
}
function marker(meta, name) { return '<!-- chatgptbox-' + meta.id + '-' + name + ' -->' }
// An accidental-edit checksum, not an authentication or secrecy boundary.
function checksum(text) {
  let first = 2166136261, second = 3339675911
  for (let index = 0; index < text.length; index++) {
    const code = text.charCodeAt(index)
    first = Math.imul(first ^ code, 16777619)
    second = Math.imul(second ^ code, 2246822519)
  }
  return text.length + ':' + (first >>> 0).toString(16) + ':' + (second >>> 0).toString(16)
}
function pendingText(pending) {
  return '## USER\n\n' + userText(pending.query) + '\n\n## ASSISTANT (pending)\n\nUse Gemini Get to collect this exact request.'
}
function span(meta, body, name) {
  const start = marker(meta, name + ':start'), end = marker(meta, name + ':end')
  if (body.split(start).length !== 2 || body.split(end).length !== 2 || body.indexOf(end) < body.indexOf(start))
    throw new Error('Damaged or duplicate Gemini controls. The complete note and original request were kept.')
  return { start, end, from: body.indexOf(start) + start.length, to: body.indexOf(end), text: body.slice(body.indexOf(start) + start.length, body.indexOf(end)).replace(/^\n|\n$/g, '') }
}
function validate(meta, body) {
  if (meta.kind !== 'conversation') return
  span(meta, body, 'waiting')
  span(meta, body, 'history')
  if (meta.pending) {
    if (!/^drafts-[a-z0-9]+-[a-z0-9]+$/.test(meta.pending.key) || !meta.pending.request || typeof meta.pending.query !== 'string')
      throw new Error('The original Gemini request is missing. Keep this note for inspection.')
    if (span(meta, body, 'pending').text !== pendingText(meta.pending))
      throw new Error('The pending text was edited. Keep the complete note and restore that generated block before Get; put new text in Waiting Reply or outside the history block.')
  } else if (['pending:start', 'pending:end'].some((name) => body.includes(marker(meta, name)))) throw new Error('Conflicting Gemini pending controls. The complete note was kept.')
}
function assertManagedText(meta, body) {
  if (meta.kind === 'conversation' && meta.historyChecksum !== checksum(span(meta, body, 'history').text))
    throw new Error('The generated history was edited. The complete note was kept. Move your notes outside its history markers, or use List/Get in a new note before refreshing.')
}
function save(meta, body) {
  validate(meta, body)
  if (meta.kind === 'conversation') meta.historyChecksum = checksum(span(meta, body, 'history').text)
  draft.content = '<!-- chatgptbox-document:' + encode(meta) + ' -->\n' + body
  draft.update()
}
function waitingBlock(meta, value = '') {
  return '\n\n## Waiting Reply\n' + marker(meta, 'waiting:start') + '\n' + value + '\n' + marker(meta, 'waiting:end') + '\n'
}
function transcript(detail) {
  if (!Array.isArray(detail.messages)) throw new Error('Gemini returned no readable conversation messages.')
  return detail.messages.map((message) => message.role === 'user'
    ? '## USER\n\n' + userText(message.content)
    : '## ASSISTANT' + (message.completed ? '' : ' (incomplete)') + '\n\n' +
      (message.supported === false ? 'This native content is not supported. Inspect the original Gemini conversation.' : message.content || 'No confirmed final text is available.')
  ).join('\n\n')
}
function detailFor(meta, cursor) {
  let ref = meta.conversationRef
  if (!ref) {
    if (!meta.nativeConversationId) throw new Error('No native conversation identity is available. Keep this note and inspect its original run.')
    const page = request('/engines/' + ENGINE + '/conversations?conversationId=' + encodeURIComponent(meta.nativeConversationId))
    const matches = page.items?.filter((item) => item.id === meta.nativeConversationId)
    if (matches?.length !== 1 || !matches[0].conversationRef) throw new Error('The original native Gemini conversation could not be identified.')
    ref = matches[0].conversationRef
  }
  return request('/engines/' + ENGINE + '/conversations/' + encodeURIComponent(ref) + '?limit=' + HISTORY_PAGE_SIZE + (cursor ? '&cursor=' + encodeURIComponent(cursor) : ''))
}
function list(meta, body) {
  if (meta.kind === 'conversation') throw new Error('Run Gemini List in a new note; this conversation and any waiting request were kept.')
  const continuing = meta.kind === 'list' && !!meta.nextCursor
  const page = request('/engines/' + ENGINE + '/conversations?limit=50' + (continuing ? '&cursor=' + encodeURIComponent(meta.nextCursor) : ''))
  if (!Array.isArray(page.items)) throw new Error('The gateway returned an invalid conversation list.')
  const items = continuing ? [...(meta.items || [])] : []
  for (const item of page.items) {
    if (typeof item.conversationRef !== 'string' || !/^[A-Za-z0-9-]+$/.test(item.conversationRef)) throw new Error('Invalid conversation reference.')
    if (!items.some((old) => old.ref === item.conversationRef)) items.push({ ref: item.conversationRef, title: safeTitle(item.title) })
  }
  if (items.length > 1000) throw new Error('This list reached 1,000 conversations. Keep it and open another note for further browsing.')
  const next = { version: 3, engineId: ENGINE, id: meta.kind === 'list' ? meta.id : identifier(), kind: 'list', items, nextCursor: page.nextCursor || null }
  const rows = items.map((item) => '- [ ] ' + item.title + ' <!-- chatgptbox-' + next.id + '-conversation:' + item.ref + ' -->')
  save(next, '# Gemini conversations\n\n' + (next.nextCursor ? 'Run List again to load more. ' : '') + 'Check one conversation, then run Gemini Get.\n\n' + rows.join('\n') + '\n')
  void body
}
function historyBlock(meta, content) { return marker(meta,'history:start') + '\n' + content + '\n' + marker(meta,'history:end') }
function unsent(meta, body, pending, message) {
  const current = span(meta, body, 'pending')
  body = body.slice(0, body.indexOf(current.start)) + '## Unsent request\n\n' + userText(pending.query) + '\n\n' + userText(message) + body.slice(current.to + current.end.length)
  meta.lastUnsent = { key: pending.key, message }
  delete meta.pending
  save(meta, body)
}
function get(meta, body, older = false) {
  validate(meta, body)
  if (meta.kind === 'list') {
    const selected = []
    const seen = new Set()
    for (const line of body.split('\n')) {
      if (!line.includes('<!-- chatgptbox-' + meta.id + '-conversation:')) continue
      const match = line.match(/^- \[([ xX])\] .* <!-- chatgptbox-([a-z0-9]+)-conversation:([A-Za-z0-9-]+) -->$/)
      if (!match || match[2] !== meta.id || seen.has(match[3]) || !meta.items?.some((item) => item.ref === match[3])) throw new Error('Damaged Gemini list. Keep the complete note.')
      seen.add(match[3])
      if (match[1].toLowerCase() === 'x') selected.push(meta.items.find((item) => item.ref === match[3]))
    }
    if (seen.size !== meta.items?.length || selected.length !== 1) throw new Error('Check exactly one Gemini conversation, then run Get.')
    meta = { version: 3, engineId: ENGINE, id: identifier(), kind: 'conversation', title: selected[0].title, conversationRef: selected[0].ref }
    body = '# ' + meta.title + '\n\n' + historyBlock(meta, '') + waitingBlock(meta)
  }
  if (meta.kind !== 'conversation') throw new Error('Use Gemini List/Get for history, or Gemini Send on an ordinary note.')
  if (older && meta.pending) throw new Error('Collect the pending request with Gemini Get before loading older messages.')
  if (meta.pending) {
    const pending = meta.pending
    if (!pending.runId) {
      const operation = request('/operations/' + encodeURIComponent(pending.key))
      pending.runId = operation.result?.runId || operation.operationId
      if (!pending.runId) throw new Error('Submission is still unconfirmed. The original request was kept; no resubmission occurred.')
      save(meta, body)
    }
    let run = request('/runs/' + encodeURIComponent(pending.runId))
    if (run.engineId !== ENGINE || run.runId !== pending.runId) throw new Error('The returned run does not match this Gemini request. The original note was kept.')
    if (!run.result && run.resultAvailability === 'recoverable' && run.phase === 'settled') {
      run = request('/runs/' + encodeURIComponent(pending.runId) + '/recover', 'POST', {})
      if (run.engineId !== ENGINE || run.runId !== pending.runId) throw new Error('The recovered run does not match this Gemini request. The original note was kept.')
    }
    if (run.phase === 'settled' && run.outcome === 'completed' && run.result?.content) {
      if (run.result.version !== 1 || !['safe-html', 'markdown'].includes(run.result.format)) throw new Error('Unsupported result format; the original run was kept.')
      const current = span(meta, body, 'pending')
      body = body.slice(0, body.indexOf(current.start)) + '## USER\n\n' + userText(pending.query) + '\n\n## ASSISTANT\n\n' + run.result.content + body.slice(current.to + current.end.length)
      meta.conversationRef = run.conversationRef || null
      meta.revision = run.conversationRevision || null
      meta.nativeConversationId = run.result.metadata?.conversationId || meta.nativeConversationId
      meta.serverHead = run.result.metadata?.serverTurnId || null
      delete meta.pending
      save(meta, body)
      app.displaySuccessMessage('Gemini answer collected. Waiting Reply was preserved.')
    } else if (run.phase === 'settled' && run.submission === 'not_attempted') {
      unsent(meta, body, pending, run.error?.message || 'The request was not submitted.')
      throw new Error('Gemini did not submit this request. Its text was retained above; edit Waiting Reply before another explicit Send.')
    } else {
      save(meta, body)
      app.displaySuccessMessage('Still waiting or unconfirmed. Use Get again; no new question was sent.')
    }
    return
  }
  if (older && !meta.historyCursor) { app.displaySuccessMessage('All available Gemini turns in this conversation have been loaded.'); return }
  const detail = detailFor(meta, older ? meta.historyCursor : undefined)
  if (older && meta.serverHead && detail.serverHead !== meta.serverHead)
    throw new Error('The native conversation changed. Use Gemini Get to refresh it before loading older turns.')
  meta.conversationRef = detail.conversationRef
  meta.revision = detail.conversationRevision
  meta.nativeConversationId = detail.conversationId
  meta.serverHead = detail.serverHead
  meta.historyCursor = detail.nextCursor || null
  if (older) {
    const current = span(meta, body, 'history')
    save(meta, body.slice(0,current.from) + '\n' + transcript(detail) + '\n\n' + current.text + '\n' + body.slice(current.to))
  } else {
    const current = span(meta, body, 'history')
    save(meta, body.slice(0,current.from) + '\n' + transcript(detail) + '\n' + body.slice(current.to))
  }
  if (meta.historyCursor) app.displaySuccessMessage('More historical turns are available with Gemini Load Older Turns.')
}
function send(meta, body) {
  if (meta.kind === 'list') throw new Error('Open the checked conversation with Gemini Get before sending. Lists are never submitted.')
  validate(meta, body)
  if (meta.pending) return get(meta, body)
  const query = meta.kind === 'new' ? body.trim() : span(meta, body, 'waiting').text.trim()
  if (!query) return get(meta, body)
  if (query.length > 50000) throw new Error('Gemini accepts up to 50,000 characters. The note was kept.')
  const continuing = meta.kind === 'conversation' && !!(meta.conversationRef || meta.nativeConversationId)
  if (continuing && (!meta.conversationRef || !Number.isInteger(meta.revision))) {
    const detail = detailFor(meta)
    if (meta.serverHead && detail.serverHead !== meta.serverHead) throw new Error('The native conversation changed. Use Get to inspect it before sending.')
    meta.conversationRef = detail.conversationRef
    meta.revision = detail.conversationRevision
  }
  const payload = { contractVersion: 1, operation: continuing ? 'conversation.append' : 'generation.start', input: { prompt: query } }
  if (continuing) payload.conversation = { ref: meta.conversationRef, expectedRevision: meta.revision }
  const key = 'drafts-' + Date.now().toString(36) + '-' + identifier()
  const pending = { key, query, request: clone(payload), createdAt: Date.now(), runId: null }
  if (meta.kind === 'new') { meta.kind = 'conversation'; meta.title = 'Gemini conversation'; body = '# Gemini conversation\n\n' + historyBlock(meta, '') + waitingBlock(meta) }
  else { const current = span(meta, body, 'waiting'); body = body.slice(0,current.from) + '\n\n' + body.slice(current.to) }
  const insertionAt = span(meta, body, 'history').to
  body = body.slice(0,insertionAt) + '\n\n' + marker(meta,'pending:start') + '\n' + pendingText(pending) + '\n' + marker(meta,'pending:end') + '\n' + body.slice(insertionAt)
  meta.pending = pending
  save(meta, body) // Complete original request is durable BEFORE the HTTP write.
  try {
    const receipt = request('/engines/' + ENGINE + '/runs', 'POST', pending.request, key)
    if (receipt.engineId !== ENGINE || typeof receipt.runId !== 'string') throw new Error('The run receipt was invalid. Keep the original request and use Get.')
    pending.runId = receipt.runId
    save(meta, body)
    app.displaySuccessMessage('Gemini request registered. Use Get; registration is not a final answer.')
  } catch (error) {
    if (error.detail?.error?.submission === 'not_attempted' || error.detail?.error?.dispatched === false) {
      unsent(meta, body, pending, error.message)
    }
    throw error
  }
}
try {
  const { meta, body } = readDocument()
  validate(meta, body)
  assertManagedText(meta, body)
  if (ACTION === 'list') list(meta, body)
  else if (ACTION === 'get') get(meta, body)
  else if (ACTION === 'send') send(meta, body)
  else if (ACTION === 'older') get(meta, body, true)
  else throw new Error('Unknown Gemini action.')
} catch (error) {
  app.displayErrorMessage(error.message || String(error))
  throw error
}
