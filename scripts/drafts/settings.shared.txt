// Nonsecret preferences have one owner outside the replaceable action packages.
const SETTINGS_PATH = '/ChatGPTBox/settings.json'
const SETTINGS_STAGE_PATH = '/ChatGPTBox/settings.pending.json'

function defaultPreferences() {
  return { version: 1, baseUrl: null, modelOverride: null, includeThinking: false }
}

function settingsSnapshot(fm) {
  const exists = fm.exists(SETTINGS_PATH)
  if (fm.lastError) throw new Error('无法读取共享设置。请确认 iCloud Drive 可用并下载 ChatGPTBox/settings.json 后重试。')
  if (!exists) return null
  const text = fm.readString(SETTINGS_PATH)
  if (typeof text !== 'string') throw new Error('无法读取共享设置。请先下载 iCloud Drive/Drafts/ChatGPTBox/settings.json 后重试。')
  return text
}

function validatePreferences(value) {
  if (!value || Array.isArray(value) || value.version !== 1 ||
      Object.keys(value).some((key) => !['version', 'baseUrl', 'modelOverride', 'includeThinking'].includes(key)) ||
      (value.baseUrl !== null && typeof value.baseUrl !== 'string') ||
      (value.modelOverride !== null && typeof value.modelOverride !== 'string') ||
      typeof value.includeThinking !== 'boolean')
    throw new Error('共享设置格式不受支持。请通过 More → 设置 检查或重新设置。')
  const baseUrl = value.baseUrl?.trim().replace(/\/+$/, '') || null
  const modelOverride = value.modelOverride?.trim() || null
  if (baseUrl && (baseUrl.length > 2048 || !/^https?:\/\/[^/?#\s@]+(?:\/[^?#\s]*)?$/i.test(baseUrl)))
    throw new Error('网关地址须为 http:// 或 https:// 地址，不能包含用户名、密码、查询参数或片段。')
  if (modelOverride && (modelOverride.length > 200 || /\s/.test(modelOverride)))
    throw new Error('模型须填写不含空格的 model slug；留空即可跟随扩展或会话。')
  return { version: 1, baseUrl, modelOverride, includeThinking: value.includeThinking }
}

function preferencesFromSnapshot(text) {
  if (text === null) return defaultPreferences()
  if (typeof text !== 'string') throw new Error('无法读取共享设置。请确认 iCloud Drive 文件可用后重试。')
  if (text.length > 16384) throw new Error('共享设置文件过大。请通过 More → 设置 检查或重新设置。')
  let value
  try { value = JSON.parse(text) }
  catch { throw new Error('共享设置不是有效 JSON。请通过 More → 设置 检查或重新设置。') }
  return validatePreferences(value)
}

function readPreferences() {
  const text = settingsSnapshot(FileManager.createCloud())
  if (text === null)
    throw new Error('此设备未找到共享设置。首次使用请先运行 More → 设置 并保存；若已在其他设备配置，请等待 iCloud Drive/Drafts/ChatGPTBox/settings.json 同步并下载后重试。当前笔记已保留，未发出请求。')
  return preferencesFromSnapshot(text)
}

function savePreferences(value, original) {
  const fm = FileManager.createCloud()
  if (settingsSnapshot(fm) !== original)
    throw new Error('共享设置已被其他操作修改。请重新打开 More → 设置 后再保存。')
  if (!fm.exists('/ChatGPTBox') && !fm.createDirectory('ChatGPTBox', '/'))
    throw new Error('无法创建共享设置目录。请确认 iCloud Drive 可写后重试。')
  if (!fm.writeJSON(SETTINGS_STAGE_PATH, value))
    throw new Error('共享设置暂存失败；未替换原设置。请确认 iCloud Drive 可写后重试。')
  const staged = fm.readString(SETTINGS_STAGE_PATH)
  if (typeof staged !== 'string' ||
      JSON.stringify(preferencesFromSnapshot(staged)) !== JSON.stringify(value))
    throw new Error('共享设置暂存失败；未替换原设置。请确认 iCloud Drive 可写后重试。')
  if (settingsSnapshot(fm) !== original)
    throw new Error('共享设置已被其他操作修改。请重新打开 More → 设置 后再保存。')
  // Native Drafts can reject moveItem/copyItem when the destination exists,
  // even with overwrite=true. Write the verified staged text, then read it back.
  if (!fm.writeString(SETTINGS_PATH, staged))
    throw new Error('无法确认共享设置已保存。请重新打开 More → 设置 检查后重试。')
  const saved = settingsSnapshot(fm)
  if (saved === null || JSON.stringify(preferencesFromSnapshot(saved)) !== JSON.stringify(value))
    throw new Error('无法确认共享设置已保存。请重新打开 More → 设置 检查后重试。')
}

function editPreferences() {
  const original = settingsSnapshot(FileManager.createCloud())
  let value
  try { value = preferencesFromSnapshot(original) }
  catch (error) {
    const recovery = Prompt.create()
    recovery.title = 'ChatGPTBox 设置需要修复'
    recovery.message = error.message + '\n请先确认文件已完成同步。重新设置后，只有点击保存才会替换原设置；API token 和当前笔记不受影响。'
    recovery.addButton('重新设置')
    if (!recovery.show()) return
    value = defaultPreferences()
  }
  let fields = { baseUrl: value.baseUrl || '', modelOverride: value.modelOverride || '', includeThinking: value.includeThinking }
  for (;;) {
    const prompt = Prompt.create()
    prompt.title = 'ChatGPTBox 设置'
    prompt.message = '所有动作共用。地址留空沿用共享凭据中的地址；模型留空跟随扩展或会话。\n模型和思考显示仅用于 ChatGPT；显示思考不会改变思考强度。更换地址后，已有笔记仍绑定原网关。'
    const textOptions = { autocapitalization: 'none', autocorrect: false }
    prompt.addTextField('baseUrl', '网关地址（留空沿用原地址）', fields.baseUrl, { ...textOptions, keyboard: 'URL' })
    prompt.addTextField('modelOverride', 'ChatGPT 模型（留空自动）', fields.modelOverride, textOptions)
    prompt.addSwitch('includeThinking', 'ChatGPT：显示思考内容', fields.includeThinking)
    prompt.addButton('保存')
    prompt.addButton('API token 设置说明')
    if (!prompt.show()) return
    fields = { ...prompt.fieldValues }
    if (prompt.buttonPressed === 'API token 设置说明') {
      const help = Prompt.create()
      help.title = 'ChatGPTBox API token'
      help.message = '所有动作沿用 Drafts Settings → Credentials → ChatGPTBox Gateway。首次网络操作会询问地址和 API token。\n更新 token 时，在该凭据项使用 Forget，再运行网络动作重新填写。此处保存的网关地址会优先使用。地址须能从每台设备访问；iPhone 的 127.0.0.1 指向 iPhone。'
      help.addButton('返回设置')
      help.show()
      continue
    }
    if (prompt.buttonPressed !== '保存') throw new Error('未知设置操作。未保存设置。')
    try { value = validatePreferences({ version: 1, ...fields }) }
    catch (error) { app.displayErrorMessage(error.message); continue }
    savePreferences(value, original)
    app.displaySuccessMessage('ChatGPTBox 设置已保存，所有动作将共用。')
    return
  }
}
